THOMAS BUILDS — SUPABASE SETUP (one time)

A) Create project
1) Go to https://supabase.com → Sign in → New project
2) Pick a name, database password (save it), region → Create
3) Wait until green. Open Project Settings → API
4) Copy "Project URL" into site-config.js → supabaseUrl
5) Copy "anon public" key into site-config.js → supabaseAnonKey
6) Authentication → Providers → enable "Anonymous sign-ins" (ON)

B) SQL Editor → New query → paste ALL of the block below → Run

-----BEGIN SQL-----
create table if not exists public.profiles (
  id uuid primary key references auth.users (id) on delete cascade,
  is_owner boolean not null default false,
  created_at timestamptz not null default now()
);

create table if not exists public.threads (
  id uuid primary key default gen_random_uuid(),
  visitor_id uuid not null references auth.users (id) on delete cascade,
  created_at timestamptz not null default now()
);

create index if not exists threads_visitor_id_idx on public.threads (visitor_id);
create unique index if not exists threads_one_visitor on public.threads (visitor_id);

create table if not exists public.messages (
  id uuid primary key default gen_random_uuid(),
  thread_id uuid not null references public.threads (id) on delete cascade,
  author_id uuid not null references auth.users (id) on delete cascade,
  body text not null default '',
  attachments jsonb not null default '[]'::jsonb,
  created_at timestamptz not null default now()
);

create index if not exists messages_thread_id_idx on public.messages (thread_id);

alter table public.profiles enable row level security;
alter table public.threads enable row level security;
alter table public.messages enable row level security;

create policy "profiles self read" on public.profiles for select using (id = auth.uid());

create policy "threads visitor insert" on public.threads for insert
with check (visitor_id = auth.uid());

create policy "threads visitor select own" on public.threads for select
using (visitor_id = auth.uid());

create policy "threads owner select all" on public.threads for select
using (exists (select 1 from public.profiles p where p.id = auth.uid() and p.is_owner = true));

create policy "messages select thread access" on public.messages for select
using (
  exists (
    select 1 from public.threads t
    where t.id = messages.thread_id
      and (t.visitor_id = auth.uid() or exists (select 1 from public.profiles p where p.id = auth.uid() and p.is_owner = true))
  )
);

create policy "messages visitor insert" on public.messages for insert
with check (
  author_id = auth.uid()
  and exists (select 1 from public.threads t where t.id = thread_id and t.visitor_id = auth.uid())
);

create policy "messages owner insert" on public.messages for insert
with check (
  author_id = auth.uid()
  and exists (select 1 from public.profiles p where p.id = auth.uid() and p.is_owner = true)
  and exists (select 1 from public.threads t where t.id = thread_id)
);

insert into storage.buckets (id, name, public)
values ('chat-media', 'chat-media', false)
on conflict (id) do nothing;

create policy "chat read own or owner" on storage.objects for select
using (
  bucket_id = 'chat-media'
  and (
    (storage.foldername(name))[1] = auth.uid()::text
    or exists (select 1 from public.profiles p where p.id = auth.uid() and p.is_owner = true)
  )
);

create policy "chat insert own prefix" on storage.objects for insert
with check (
  bucket_id = 'chat-media'
  and (storage.foldername(name))[1] = auth.uid()::text
);

create policy "chat delete own prefix" on storage.objects for delete
using (
  bucket_id = 'chat-media'
  and (storage.foldername(name))[1] = auth.uid()::text
);

alter publication supabase_realtime add table public.messages;
-----END SQL-----

If the last line errors with "already member of publication", ignore it (realtime is already on).

Optional: turn off the landing background clip by setting heroVideoUrl to "" in site-config.js.

C) Make yourself the "owner" (so admin.html can see every thread)
1) Authentication → Users → Add user → your real email, set a password (or use Magic link later from admin page)
2) SQL Editor → run (replace the email):

   update public.profiles set is_owner = true
   where id = (select id from auth.users where email = 'YOUR_EMAIL_HERE');

   If profiles row missing for that user, run once:

   insert into public.profiles (id, is_owner)
   select id, true from auth.users where email = 'YOUR_EMAIL_HERE'
   on conflict (id) do update set is_owner = excluded.is_owner;

D) Auth URLs for owner login (main site Inbox + optional admin.html)
Authentication → URL configuration
Site URL: https://YOUR-SITE.workers.dev/ (or your custom domain root)
Redirect URLs: add the same root URL, and if you still use admin.html add https://YOUR-SITE.workers.dev/admin.html
Magic links return to Site URL with a hash — then scroll to “Inbox (Thomas only)” on the homepage.

E) Redeploy / refresh the static site after editing site-config.js on GitHub.

Notes
- Visitors stay anonymous; each browser gets one Supabase anonymous user and one thread.
- You answer from admin.html after signing in with your owner email (magic link or password, depending on what you enabled).
